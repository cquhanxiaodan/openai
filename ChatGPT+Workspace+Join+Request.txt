  function getViewportPanelWidth() {
    return Math.max(320, Math.min(Number(CONFIG.panelWidth) || DEFAULTS.panelWidth, window.innerWidth - 16));
  }

  function getDefaultPanelLeft() {
    return Math.max(8, window.innerWidth - getViewportPanelWidth() - 14);
  }

  function normalizePanelPosition(left, top) {
    const width = panelEl?.offsetWidth || getViewportPanelWidth();
    const height = panelEl?.offsetHeight || 0;
    const maxLeft = Math.max(8, window.innerWidth - width - 8);
    const maxTop = Math.max(8, window.innerHeight - Math.max(height, 120) - 8);
    const safeLeft = Number.isFinite(left) ? Math.min(Math.max(8, left), maxLeft) : getDefaultPanelLeft();
    const safeTop = Number.isFinite(top) ? Math.min(Math.max(8, top), maxTop) : Number(DEFAULTS.panelTop);
    return {
      left: Math.round(safeLeft),
      top: Math.round(safeTop),
    };
  }

  function applyPanelPosition() {
    if (!panelEl) return;
    const rawLeft = CONFIG.panelLeft === null || CONFIG.panelLeft === "" ? NaN : Number(CONFIG.panelLeft);
    const rawTop = CONFIG.panelTop === null || CONFIG.panelTop === "" ? NaN : Number(CONFIG.panelTop);
    const pos = normalizePanelPosition(rawLeft, rawTop);
    panelEl.style.left = `${pos.left}px`;
    panelEl.style.top = `${pos.top}px`;
    panelEl.style.right = "auto";
    CONFIG.panelLeft = pos.left;
    CONFIG.panelTop = pos.top;
  }

  function setPanelCollapsed(collapsed, persist = true) {
    CONFIG.panelCollapsed = !!collapsed;
    if (panelEl) panelEl.classList.toggle("jr-panel-collapsed", CONFIG.panelCollapsed);
    if (panelConfigWrapEl) panelConfigWrapEl.setAttribute("aria-hidden", CONFIG.panelCollapsed ? "true" : "false");
    if (persist) saveConfig(CONFIG);
  }

  function togglePanelCollapsed() {
    setPanelCollapsed(!CONFIG.panelCollapsed);
  }

  function initPanelInteractions() {
    if (!panelEl || !panelHeadEl) return;
    let dragState = null;

    const finishDrag = (e) => {
      if (!dragState || e.pointerId !== dragState.pointerId) return;
      const moved = dragState.moved;
      try { panelHeadEl.releasePointerCapture?.(e.pointerId); } catch (_) {}
      panelHeadEl.classList.remove("jr-head-dragging");
      dragState = null;
      if (moved) {
        const rect = panelEl.getBoundingClientRect();
        const pos = normalizePanelPosition(rect.left, rect.top);
        CONFIG.panelLeft = pos.left;
        CONFIG.panelTop = pos.top;
        applyPanelPosition();
        saveConfig(CONFIG);
      } else {
        togglePanelCollapsed();
      }
    };

    panelHeadEl.addEventListener("pointerdown", (e) => {
      if (e.button !== 0) return;
      const rect = panelEl.getBoundingClientRect();
      dragState = {
        pointerId: e.pointerId,
        startX: e.clientX,
        startY: e.clientY,
        startLeft: rect.left,
        startTop: rect.top,
        moved: false,
      };
      panelHeadEl.classList.add("jr-head-dragging");
      try { panelHeadEl.setPointerCapture?.(e.pointerId); } catch (_) {}
    });

    panelHeadEl.addEventListener("pointermove", (e) => {
      if (!dragState || e.pointerId !== dragState.pointerId) return;
      const dx = e.clientX - dragState.startX;
      const dy = e.clientY - dragState.startY;
      if (!dragState.moved && (Math.abs(dx) > 3 || Math.abs(dy) > 3)) dragState.moved = true;
      if (!dragState.moved) return;
      e.preventDefault();
      const pos = normalizePanelPosition(dragState.startLeft + dx, dragState.startTop + dy);
      panelEl.style.left = `${pos.left}px`;
      panelEl.style.top = `${pos.top}px`;
      panelEl.style.right = "auto";
    });

    panelHeadEl.addEventListener("pointerup", finishDrag);
    panelHeadEl.addEventListener("pointercancel", finishDrag);

    window.addEventListener("resize", () => {
      if (!panelEl) return;
      applyPanelPosition();
      saveConfig(CONFIG);
    });
  }

  function pickFirstNonEmpty(...values) {
    for (const value of values) {
      if (value === undefined || value === null) continue;
      const text = String(value).trim();
      if (text) return text;
    }
    return "";
  }

  function decodeJwt(at) {
    try {
      const p = at.split(".")[1];
      const j = JSON.parse(atob(p.replace(/-/g, "+").replace(/_/g, "/")));
      const auth = j["https://api.openai.com/auth"] || {};
      const prof = j["https://api.openai.com/profile"] || {};
      return {
        account_id: auth.chatgpt_account_id || "",
        chatgpt_user_id: auth.chatgpt_user_id || auth.user_id || "",
        email: prof.email || "",
        plan_type: auth.chatgpt_plan_type || "",
        client_id: j.client_id || "",
        exp: j.exp || 0,
      };
    } catch (_) {
      return {};
    }
  }

  function extractSessionInfo(session, at) {
    const jwt = at ? decodeJwt(at) : {};
    const account = session?.account || {};
    const user = session?.user || {};
    return {
      account_id: jwt.account_id || account.id || "",
      workspace_id: account.id || jwt.account_id || "",
      email: user.email || jwt.email || "",
      plan_type: account.planType || jwt.plan_type || "",
      exp: jwt.exp || 0,
      chatgpt_user_id: jwt.chatgpt_user_id || user.id || "",
      organization_id: account.organizationId || "",
      client_id: jwt.client_id || "",
      session_token: session?.sessionToken || session?.session_token || "",
    };
  }

    if (!exp) return "?";
    const min = Math.round((exp * 1000 - Date.now()) / 60000);
    if (min > 60) return `剩余 ${Math.round(min / 60)} 小时`;
    return `剩余 ${min} 分钟`;

    const text = String(value || "").trim();
    if (!text) return null;
    const num = Number(text);
    if (!Number.isFinite(num) || num <= 0) return null;
    return Math.trunc(num);
  }

    const token = buildSub2ApiKeyValue();
    const headers = {
      Accept: "application/json, text/plain, */*",
      "Content-Type": "application/json",

    if (token) {
      headers.Authorization = `Bearer ${token}`;
      headers["X-API-Key"] = token;

    if (base) {

      headers.Referer = `${base}/admin/accounts`;
    }
    return headers;

  function gmRequest(method, url, payload, headers, timeout) {
    return new Promise((resolve, reject) => {
      if (typeof GM_xmlhttpRequest !== "function") {
        reject(new Error("GM_xmlhttpRequest 不可用，请确认脚本运行在 Tampermonkey"));
        return;
      }
      const finalHeaders = Object.assign({}, headers || {});
      let body;
      if (payload !== undefined && payload !== null) {
        body = typeof payload === "string" ? payload : JSON.stringify(payload);
        if (!finalHeaders["Content-Type"] && !finalHeaders["content-type"]) {
          finalHeaders["Content-Type"] = "application/json";
        }
      }
      GM_xmlhttpRequest({
        method,
        url,
        data: body,
        headers: finalHeaders,
        timeout: timeout || 15000,
        onload: (r) => resolve({
          ok: r.status >= 200 && r.status < 300,
          status: r.status,
          text: r.responseText || "",
          headers: r.responseHeaders || "",
        }),
        onerror: () => reject(new Error(`请求失败: ${method} ${url}`)),
        ontimeout: () => reject(new Error(`请求超时: ${method} ${url}`)),
      });
    });
  }

  async function gmJsonRequest(method, url, payload, headers, timeout) {
    const res = await gmRequest(method, url, payload, headers, timeout);
    let data = {};
    try { data = JSON.parse(res.text || "{}"); } catch (_) {}
    return Object.assign({}, res, { data });
  }

  function waitForElement(selector, options) {
    options = options || {};
    const timeoutMs = Number(options.timeoutMs || 20000);
    const root = options.root || document;
    const predicate = typeof options.predicate === "function" ? options.predicate : null;
    return new Promise((resolve, reject) => {
      const started = Date.now();
      const find = () => {
        const nodes = Array.from(root.querySelectorAll(selector));
        const hit = predicate ? nodes.find(predicate) : nodes[0];
        if (hit) return hit;
        return null;
      };
      const immediate = find();
      if (immediate) {
        resolve(immediate);
        return;
      }
      const timer = setInterval(() => {
        const node = find();
        if (node) {
          clearInterval(timer);
          resolve(node);
          return;
        }
        if (Date.now() - started >= timeoutMs) {
          clearInterval(timer);
          reject(new Error(`等待元素超时: ${selector}`));
        }
      }, 250);
    });
  }

    el.dispatchEvent(new MouseEvent("mouseover", { bubbles: true, cancelable: true, view: window }));
    el.dispatchEvent(new MouseEvent("mousedown", { bubbles: true, cancelable: true, view: window }));
    el.dispatchEvent(new MouseEvent("mouseup", { bubbles: true, cancelable: true, view: window }));
    el.click();
    return true;
  }

  async function copyText(text) {
    if (navigator.clipboard && navigator.clipboard.writeText) {
      await navigator.clipboard.writeText(text);
      return;
    }
    const ta = document.createElement("textarea");
    ta.value = text;
    ta.setAttribute("readonly", "");
    ta.style.position = "fixed";
    ta.style.opacity = "0";
    ta.style.pointerEvents = "none";
    document.body.appendChild(ta);
    ta.select();
    const ok = document.execCommand("copy");
    ta.remove();
    if (!ok) throw new Error("execCommand copy failed");
  }

    const oldBg = btn.style.background;
    const oldColor = btn.style.color;

    btn.style.background = bg;
    btn.style.color = "#fff";
    setTimeout(() => {
      if (!btn.isConnected) return;
      btn.textContent = oldText;
      btn.style.background = oldBg;
      btn.style.color = oldColor;
    }, 1200);
  }

  async function checkLocalSub2Server() {
    const base = getLocalServerBase();
    if (!base) throw new Error("未配置本地回调服务地址");
    const result = await gmJsonRequest("GET", `${base}/health?_t=${Date.now()}`, null, {}, 8000);
    if (!result.ok || !result.data?.ok) {
      throw new Error(`本地 callback server 不可用: HTTP ${result.status}`);

  }

  async function saveSub2TokensViaLocalServer(email, data) {
    const base = getLocalServerBase();
    const normalizedEmail = String(data?.email || email || "").trim().toLowerCase();
    if (!base) throw new Error("未配置本地回调服务地址");
    if (!normalizedEmail) throw new Error("缺少邮箱，无法保存 token");
    const result = await gmJsonRequest("POST", `${base}/save-token`, {
      email: normalizedEmail,
      notes: "",
      data,
    }, {}, 15000);
    if (!result.ok) throw new Error(`保存 token 失败 HTTP ${result.status}${result.text ? `: ${result.text}` : ""}`);
    if (!result.data?.ok || !result.data?.cpa_file || !result.data?.sub2_file) {
      throw new Error(`保存 token 返回异常: ${result.text || "empty"}`);
    }
    return result.data;
  }

  async function registerPendingSub2PushWithLocalServer(payload) {
    const base = getLocalServerBase();
    if (!base) throw new Error("未配置本地回调服务地址");
    const result = await gmJsonRequest("POST", `${base}/sub2/pending`, payload || {}, {}, 15000);
    if (!result.ok) throw new Error(`登记 Sub2 pending 失败 HTTP ${result.status}${result.text ? `: ${result.text}` : ""}`);
    if (!result.data?.ok || !result.data?.job?.state) {
      throw new Error(`登记 Sub2 pending 返回异常: ${result.text || "empty"}`);
    }
    return result.data.job;
  }

  async function queryPendingSub2JobFromLocalServer(state) {
    const base = getLocalServerBase();
    const normalizedState = String(state || "").trim();
    if (!base || !normalizedState) return null;
    const result = await gmJsonRequest("GET", `${base}/sub2/status?state=${encodeURIComponent(normalizedState)}&_t=${Date.now()}`, null, {}, 8000);
    if (!result.ok) throw new Error(`查询 Sub2 状态失败 HTTP ${result.status}${result.text ? `: ${result.text}` : ""}`);
    return result.data?.job || null;
  }

  async function createSub2GenerateAuth(proxyId) {
    const base = normalizeSub2ApiBase(CONFIG.sub2ApiUrl);
    if (!base || !buildSub2ApiKeyValue()) throw new Error("Sub2Api 未配置 URL 或 Admin API Key");
    const normalizedProxyId = parseOptionalSub2ProxyId(proxyId);
    const result = await gmJsonRequest("POST", `${base}/api/v1/admin/openai/generate-auth-url`, {
      proxy_id: normalizedProxyId,
    }, buildSub2AdminHeaders(base), 20000);
    if (!result.ok) throw new Error(`生成 auth_url 失败 HTTP ${result.status}${result.text ? `: ${result.text}` : ""}`);
    if (result.data?.code !== 0 || !result.data?.data?.auth_url || !result.data?.data?.session_id) {
      throw new Error(`生成 auth_url 返回异常: ${result.text || "empty"}`);
    }
    return result.data.data;
  }

  async function resolveExistingPendingSub2BeforeStart(normalizedEmail) {
    const pending = getPendingSub2();
    if (!pending?.state) return false;

    const pendingEmail = String(pending.email || "").trim().toLowerCase();
    const pendingState = String(pending.state || "").trim();
    const pendingCreatedAt = Number(pending.updated_at || pending.created_at || 0);
    const pendingAgeMs = pendingCreatedAt ? (Date.now() - pendingCreatedAt) : 0;

    try {
      const job = await queryPendingSub2JobFromLocalServer(pendingState);
      const status = String(job?.status || pending.status || "").trim().toLowerCase();

      if (status === "finished") {
        clearPendingSub2();
        log(`♻️ [sub2api] 检测到残留 finished pending，已自动清理：${job?.email || pendingEmail || normalizedEmail}`, "info");
        return false;
      }

      if (status === "failed") {
        clearPendingSub2();
        log(`♻️ [sub2api] 检测到残留 failed pending，已自动清理：${job?.error || "未知错误"}`, "warn");
        return false;
      }

      if (["await_callback", "session_selected", "consent_submitted", "exchanging"].includes(status)) {
        if (pendingAgeMs > 10 * 60 * 1000) {
          clearPendingSub2();
          log(`♻️ [sub2api] 检测到超时 pending（${Math.round(pendingAgeMs / 60000)} 分钟），已自动清理并允许重开：${pendingEmail || normalizedEmail}`, "warn");
          return false;
        }
        const resumeUrl = String(job?.auth_url || pending.auth_url || "").trim();
        if (resumeUrl) {
          log(`↪️ [sub2api] 检测到未完成 OAuth，继续跳回授权页：${job?.email || pendingEmail || normalizedEmail}`, "warn");
          window.location.replace(resumeUrl);
          return true;
        }
        log(`⏳ [sub2api] 当前仍有未完成 pending：${job?.email || pendingEmail || normalizedEmail} · ${status}`, "warn");
        return true;
      }
    } catch (e) {
      if (pendingAgeMs > 10 * 60 * 1000) {
        clearPendingSub2();
        log(`♻️ [sub2api] 本地 pending 查询失败且任务已超时，已自动清理：${e.message}`, "warn");
        return false;
      }
      log(`⚠️ [sub2api] 本地 pending 查询失败，暂按未完成处理：${e.message}`, "warn");
      if (pending.auth_url) {
        window.location.replace(pending.auth_url);
        return true;
      }
      return true;
    }

    if (!pendingState || !pendingEmail || (pendingEmail && pendingEmail !== normalizedEmail && pendingAgeMs > 2 * 60 * 1000)) {
      clearPendingSub2();
      log(`♻️ [sub2api] 检测到无效/残留 pending，已自动清理`, "warn");
      return false;
    }

    log(`⏳ [sub2api] Sub2 推送仍在进行：${pendingEmail || normalizedEmail}`, "warn");
    return true;
  }

  async function startSub2ApiPushFlow(info, options) {
    options = options || {};
    if (STATE.sub2Starting) {
      log("Sub2 推送已在进行中，忽略重复触发", "warn");
      return false;
    }
    if (!hasSub2Config()) throw new Error("未配置 Sub2Api URL/KEY");

    const normalizedEmail = String(info?.email || STATE.session?.user?.email || "").trim().toLowerCase();
    if (!normalizedEmail) throw new Error("缺少邮箱，无法推送 Sub2Api");
    if (!STATE.at) throw new Error("当前 access token 为空");

    if (await resolveExistingPendingSub2BeforeStart(normalizedEmail)) {
      return false;
    }

    const localInfo = await checkLocalSub2Server();
    const sessionData = buildSub2SessionData(STATE.session, info, STATE.at);
    const fingerprint = computeSub2Fingerprint(sessionData);
    const proxyId = parseOptionalSub2ProxyId(CONFIG.sub2ProxyId);

    STATE.sub2Starting = true;
    setSub2BtnBusy(true);
    try {
      const saved = await saveSub2TokensViaLocalServer(normalizedEmail, sessionData);
      log(`💾 [sub2api] Token 已保存：${saved.cpa_file} + ${saved.sub2_file}`, "ok");

      const authInfo = await createSub2GenerateAuth(proxyId);
      const authState = parseAuthStateFromUrl(authInfo.auth_url);
      if (!authState) throw new Error("auth_url 中未取到 state");

      const pendingPayload = {
        email: normalizedEmail,
        manual: !!options.manual,
        proxy_id: proxyId,
        session_id: authInfo.session_id,
        auth_url: authInfo.auth_url,
        state: authState,
        created_at: Date.now(),
        callback_url: (localInfo && localInfo.callback_url) || "http://localhost:1455/auth/callback",
        sub2_api_url: normalizeSub2ApiBase(CONFIG.sub2ApiUrl),
        sub2_api_key: buildSub2ApiKeyValue(),
        sub2_api_group_ids: String(CONFIG.sub2ApiGroupIds || "").trim(),
        current_url: window.location.href,
      };

      await registerPendingSub2PushWithLocalServer(pendingPayload);
      setPendingSub2(Object.assign({}, pendingPayload, {
        status: "await_callback",
        fingerprint,
      }));
      setLastSub2Fingerprint(fingerprint);

      log("✅ [sub2api] 已登记到本地 Python 回调服务", "ok");
      log(`🔗 [sub2api] 跳转 OAuth：${authInfo.auth_url}`, "info");
      window.location.replace(authInfo.auth_url);
      return true;
    } catch (e) {
      clearPendingSub2();
      throw e;
    } finally {
      STATE.sub2Starting = false;
      setSub2BtnBusy(false);
    }
  }

  function isActuallyVisible(el) {
    if (!el) return false;
    const rect = typeof el.getBoundingClientRect === "function" ? el.getBoundingClientRect() : null;
    const style = window.getComputedStyle ? window.getComputedStyle(el) : null;
    return !!rect &&
      rect.width > 0 &&
      rect.height > 0 &&
      (!style || (style.display !== "none" && style.visibility !== "hidden" && style.opacity !== "0"));
  }

  function shouldThrottleAuthPageAction(key, cooldownMs) {
    const lock = getAuthPageActionLock();
    const now = Date.now();
    if (lock.key === key && Number(lock.ts || 0) && now - Number(lock.ts) < Number(cooldownMs || 6000)) {
      return true;
    }
    setAuthPageActionLock({ key, ts: now });
    return false;
  }

    const requiredMs = Number(minMs || 3500);

  }

  function isPendingOwnedByCurrentPage(pending, staleMs) {
    const normalizedStaleMs = Number(staleMs || 30000);

    const updatedAt = Number(pending.owner_updated_at || 0);
    if (!updatedAt) return true;
    return Date.now() - updatedAt >= normalizedStaleMs;
  }

  async function activateAuthElement(el, options) {
    options = options || {};
    if (!el) return false;
    const form = options.form || el.form || (typeof el.closest === "function" ? el.closest("form") : null);
    const shouldFlushSubmit = !!options.preferSubmit || !!(typeof el.matches === "function" && el.matches('button[type="submit"], input[type="submit"], button[name="session_id"]'));
    try { el.scrollIntoView?.({ block: "center", inline: "center", behavior: "instant" }); } catch (_) {}
    try { el.focus?.(); } catch (_) {}
    await sleep(Number(options.preDelayMs || 350));
    try {
      if (options.preferSubmit && form && typeof form.requestSubmit === "function") {
        form.requestSubmit(el);
      } else {
        fireClick(el);
      }
    } catch (_) {
      fireClick(el);
    }
    if (shouldFlushSubmit) {
      await sleep(80);
      await flushPendingAuthSubmit();
    }
    await sleep(Number(options.postDelayMs || 1000));
    return true;
  }

  async function reconcilePendingSub2(reason) {
    const pending = getPendingSub2();
    if (!pending?.state) return false;
    try {
      const job = await queryPendingSub2JobFromLocalServer(pending.state);
      if (!job) return false;
      const status = String(job.status || "").trim().toLowerCase();
      const logKey = `${pending.state}:${status}`;

      if (status === "finished") {
        clearPendingSub2();
        setLastSub2Fingerprint(pending.fingerprint || getLastSub2Fingerprint());
        log(`🎉 [sub2api] Python 回调推送成功：${job.email || pending.email}${job.group_text ? ` · ${job.group_text}` : ""}`, "ok");
        if (job.cpa_file || job.sub2_file) {
          log(`💾 [sub2api] 文件：${[job.cpa_file, job.sub2_file].filter(Boolean).join(" + ")}`, "info");
        }
        return true;
      }

      if (status === "failed") {
        clearPendingSub2();
        log(`❌ [sub2api] Python 回调推送失败：${job.error || "未知错误"}`, "err");
        return true;
      }

      if (logKey !== STATE.lastSub2StatusKey) {
        STATE.lastSub2StatusKey = logKey;
        log(`⏳ [sub2api] 当前状态：${status}${reason ? ` · ${reason}` : ""}`, "info");
      }
      return false;
    } catch (e) {
      const warnKey = `err:${pending.state}:${e.message}`;
      if (warnKey !== STATE.lastSub2StatusKey) {
        STATE.lastSub2StatusKey = warnKey;
        log(`⚠️ [sub2api] 查询本地回调状态失败：${e.message}`, "warn");
      }
      return false;
    }
  }

  async function refreshSession() {
    try {
      const s = await fetchSession();
      const at = s.accessToken || "";
      const prevAt = STATE.at || "";
      const prevWorkspaceId = STATE.workspaceId || "";

      STATE.session = s;
      const info = extractSessionInfo(s, at);
      STATE.workspaceId = info.workspace_id || "";

      if (info.workspace_id && info.workspace_id !== prevWorkspaceId) {
        log(`当前 Workspace ID: ${info.workspace_id}`, "info");
      }

      if (at) {
        const tokenChanged = at !== prevAt;
        STATE.at = at;

        if (tokenChanged) {
          log(`子号 AT 已更新: ${info.email || "?"}`, "ok");
        }

        updateUserBar(info, "ok");
        await reconcilePendingSub2(tokenChanged ? "token-updated" : "session-poll");

        if (tokenChanged) await onATReady(info);
      } else {
        updateUserBar(null, "warn");
      }
    } catch (e) {
      log(`session 获取失败: ${e.message}`, "warn");
      updateUserBar(null, "err");
    }
  }

  async function sendOne(wsId, route, attempt) {
    attempt = attempt || 0;
    const url = `/backend-api/accounts/${wsId}/invites/${route}`;
    const headers = {
      accept: "*/*",
      authorization: "Bearer " + STATE.at,
      "content-type": "application/json",
      "oai-device-id": STATE.deviceId,
      "oai-language": navigator.language || "en-US",
    };
    log(`→ POST /accounts/${wsId.slice(0, 8)}/invites/${route} (第 ${attempt + 1} 次)`);
    try {
      const res = await fetch(url, {
        method: "POST",
        headers,
        body: "",
        mode: "cors",
        credentials: "include",
      });
      const text = await res.text();
      if (res.ok) {
        log(`✓ ${wsId.slice(0, 8)} HTTP ${res.status}: ${text}`, "ok");
        return true;
      }
      log(`✗ ${wsId.slice(0, 8)} HTTP ${res.status}: ${text.slice(0, 180)}`, "warn");
      if (res.status === 401 || res.status === 403) {
        log("子号 AT 失效，刷新 session...", "warn");
        STATE.at = "";
        await refreshSession();
        if (attempt < CONFIG.maxRetries) {
          await sleep(2000);
          return sendOne(wsId, route, attempt + 1);
        }
        return false;
      }
      if (attempt < CONFIG.maxRetries) {
        await sleep(CONFIG.retryBackoffMs * (attempt + 1));
        return sendOne(wsId, route, attempt + 1);
      }
      return false;
    } catch (e) {
      log(`网络错误: ${e.message}`, "err");
      if (attempt < CONFIG.maxRetries) {
        await sleep(CONFIG.retryBackoffMs);
        return sendOne(wsId, route, attempt + 1);
      }
      return false;
    }
  }

  function parseWorkspaceIds() {
    return String(CONFIG.workspaceIds || "")
      .split(/[\n,]+/)
      .map(s => s.trim())
      .filter(Boolean);
  }

  async function runAll(route) {
    if (STATE.running) { log("正在运行中，请稍候", "warn"); return; }
    if (!STATE.at) {
      log("无可用 AT，刷新 session...", "warn");
      await refreshSession();
      if (!STATE.at) { log("仍未取到 AT，请先登录 chatgpt.com", "err"); return; }
    }

    const ids = parseWorkspaceIds();
    if (!ids.length) { log("未配置 workspace ID", "err"); return; }

    STATE.running = true;
    STATE.runningRoute = route;
    syncActionButtons();
    log(`开始处理 ${ids.length} 个 workspace（${route}）`, "info");
    let ok = 0;
    try {
      for (const ws of ids) {
        const r = await sendOne(ws, route);
        if (r) ok++;
        if (ids.length > 1) await sleep(CONFIG.intervalMs);
      }
      log(`完成：成功 ${ok}/${ids.length}`, ok === ids.length ? "ok" : "warn");
    } finally {
      STATE.running = false;
      STATE.runningRoute = "";
      syncActionButtons();
    }
  }

  function onATReady(info) {
    if (!STATE.autoRan) {
      STATE.autoRan = true;
      Promise.resolve().then(() => runAll("request")).catch((e) => {
        log(`自动 request 失败: ${e.message}`, "err");
      });
    }
  }

  function sleep(ms) {
    return new Promise(r => setTimeout(r, ms));
  }

  function syncActionButtons() {
    if (reqBtnEl) {
      reqBtnEl.disabled = !!STATE.running;
      reqBtnEl.textContent = STATE.runningRoute === "request" ? "Request中" : "Request";
      reqBtnEl.style.opacity = reqBtnEl.disabled ? "0.5" : "1";
    }
    if (accBtnEl) {
      accBtnEl.disabled = !!STATE.running;
      accBtnEl.textContent = STATE.runningRoute === "accept" ? "Accept中" : "Accept";
      accBtnEl.style.opacity = accBtnEl.disabled ? "0.5" : "1";
    }
    if (sub2BtnEl) {
      sub2BtnEl.disabled = !!STATE.sub2Starting;
      sub2BtnEl.textContent = STATE.sub2Starting ? "Push中" : "手动 Push Sub2";
      sub2BtnEl.style.opacity = sub2BtnEl.disabled ? "0.5" : "1";
    }
    if (refreshBtnEl) {
      refreshBtnEl.style.opacity = "1";
    }
  }

  function setSub2BtnBusy(busy) {
    STATE.sub2Starting = !!busy;
    syncActionButtons();
  }

  function updateUserBar(info, status) {
    if (!userBarEl) return;
    const c = { ok: "#2f855a", warn: "#b7791f", err: "#c53030" };
    if (info && (info.email || info.workspace_id)) {
      const sub2State = hasSub2Config()
        ? `<span style="color:#2f855a">Sub2 已配置</span>`
        : `<span style="color:#a0aec0">Sub2 未配置</span>`;
      userBarEl.innerHTML =
        `<div style="display:flex;align-items:center;gap:6px;flex-wrap:wrap">` +
          `<span style="color:${c.ok}">●</span> <b>${info.email || "当前账号"}</b> · ${info.plan_type || "?"} · ` +
          `<span style="color:#718096">${fmtExp(info.exp)}</span> · ${sub2State}` +
        `</div>` +
        `<div style="margin-top:4px;font-size:11px;line-height:1.45;color:#4a5568;display:flex;align-items:center;gap:6px;flex-wrap:wrap">` +
          `<span>当前 Workspace ID：</span>` +
          `<code style="background:#edf2f7;padding:1px 4px;border-radius:3px;word-break:break-all">${info.workspace_id || info.account_id || "-"}</code>` +
          `<button type="button" class="jr-copy-btn" data-action="copy-current-workspace">复制</button>` +
        `</div>`;
    } else {
      const msg = status === "err" ? "session 获取失败，请确认已登录" : "未检测到 AT，等待登录...";
      userBarEl.innerHTML = `<span style="color:${c[status] || c.warn}">●</span> ${msg}`;
    }
  }

  function markDirty() {
    dirty = true;
    if (saveBtnEl) {
      saveBtnEl.textContent = "保存 *";
      saveBtnEl.style.background = "#d69e2e";
      saveBtnEl.style.color = "#fff";
    }
  }

  function markClean() {
    dirty = false;
    if (saveBtnEl) {
      saveBtnEl.textContent = "已保存";
      saveBtnEl.style.background = "#38a169";
      saveBtnEl.style.color = "#fff";
    }
    setTimeout(() => {
      if (!dirty && saveBtnEl) {
        saveBtnEl.textContent = "保存";
        saveBtnEl.style.background = "#edf2f7";
        saveBtnEl.style.color = "#4a5568";
      }
    }, 1500);
  }

  function log(msg, level) {
    const styles = { info: "jr-info", ok: "jr-ok", warn: "jr-warn", err: "jr-err" };
    const time = new Date().toLocaleTimeString();
    console.log("%c[JoinReq]", "color:#3182ce;font-weight:bold", msg);
    if (panelBody) {
      const line = document.createElement("div");
      line.className = `jr-line ${styles[level] || "jr-info"}`;
      line.textContent = `[${time}] ${msg}`;
      panelBody.appendChild(line);
      panelBody.scrollTop = panelBody.scrollHeight;
    }
  }

  function boot() {
    buildPanel();
    log("脚本已加载 v4.4.2", "info");
    log("标题栏支持拖动；单击标题栏可折叠/展开配置区", "info");
    if (isAuthOpenAiHost()) {
      lastAuthObservedUrl = location.href;
      lastAuthObservedAt = Date.now();
      if (userBarEl) {
        userBarEl.innerHTML = `<span style="color:#2b6cb0">●</span> 当前位于 OpenAI 授权页，浮窗与自动监控已启用`;
      }
      log("已启用 auth.openai.com / accounts.openai.com 持续检测", "info");
      startAuthOpenAiMonitor();
      return;
    }
    log("request = 主动申请 · accept = 接受邀请", "info");
    if (hasSub2Config()) {
      log("Sub2 已配置：点击“手动 Push Sub2”后才会推送", "info");
    } else {
      log("未配置 Sub2Api/Key：当前仅执行 Workspace request/accept", "warn");
    }
    refreshSession();
    setInterval(refreshSession, CONFIG.sessionPollMs);
    window.addEventListener("focus", () => { if (!STATE.running) refreshSession(); });
  }

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", boot);
  } else {
    boot();
  }
